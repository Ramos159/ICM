  let anchorX = mouseX;
  let anchorY = mouseY;

  // 2. The Body (Anchor)
  fill(255);
  ellipse(anchorX, anchorY, 100, 120);

  // 3. The Eyes (Plug in your worksheet offsets!)
  fill(0);
  ellipse(anchorX ___, anchorY ___, 15, 15); // Left Eye

}

function setup() {
  createCanvas(400, 400);
}

function draw() {
  background(30);
  
  // 1. The Mouse Anchor
  let anchorX = mouseX;
  let anchorY = mouseY;

  // 2. Head & Antennas base (Anchor)
  stroke(255);
  strokeWeight(4);
  
  // 3. Lines connecting antennas (Plug in offsets here too!)
  line(anchorX, anchorY, anchorX ___, anchorY ___); // Left line
  line(anchorX, anchorY, anchorX ___, anchorY ___); // Right line

  fill(100, 255, 100);
  noStroke();
  ellipse(anchorX, anchorY, 80, 80); // The Head

  // 4. The Antenna Balls (Plug in your worksheet offsets!)
  ellipse(anchorX ___, anchorY ___, 20, 20); // Left Antenna
  ellipse(anchorX ___, anchorY ___, 20, 20); // Right Antenna
}

Part 3: The Robot (No Guard Rails)

function setup() {
  createCanvas(400, 400);
  rectMode(CENTER); // This makes the rectangle anchor from the middle!
}

function draw() {
  background(30);
  
  // 1. The Mouse Anchor
  let anchorX = mouseX;
  let anchorY = mouseY;

  // 2. The Features (Wheels & Eyes)
  fill(100);
  rect(anchorX ___, anchorY ___, 30, 40); // Left Wheel
  rect(anchorX ___, anchorY ___, 30, 40); // Right Wheel

  // 3. The Body (Anchor)
  fill(200, 200, 250);
  rect(anchorX, anchorY, 120, 120);

  // 4. The Eyes
  fill(255, 0, 0);
  ellipse(anchorX ___, anchorY ___, 20, 20); // Left Eye
  ellipse(anchorX ___, anchorY ___, 20, 20); // Right Eye
}
